# Function to drop rows with null values in a specified column
drop_null_rows <- function(df, column) {
  # Drop rows where the specified column is null
  x<-length(df)
  df2 <- df[!is.na(df[[column]]) & df[[column]] != "" & df[[column]] != "NA" & df[[column]] != "N/A", ]
  y<-x-length(df2)
  print(paste("Rows dropped:",y))
  return(df2)
}
